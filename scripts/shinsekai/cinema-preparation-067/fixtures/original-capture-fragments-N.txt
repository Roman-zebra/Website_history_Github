async function capture() {
  requirePreparedCapture067();
  if (controller?.needsTick) controller.pause();
  viewportOwner?.sync();
  applyPose();
  observeRender067(renderer, scene, camera, {site:"original-capture", sourceFrames:frames, ready, disposed, bootDone, cameraId, ownedModels:models.size, hallState:hall.snapshot().state, exteriorState:exterior.snapshot().state, cinemaState:cinema.snapshot().state});
  const receipt = snapshot2(), blob = await new Promise((r) => canvas.toBlob(r, "image/png"));
  if (!blob || disposed) throw Error("Capture canceled");
  const captureId = staticRecorder.downloadPNG(blob, receipt);
  $("status").textContent = "PNG download requested. Matching receipt is retained below.";
  event("capture", { captureId });
  ui();
}
